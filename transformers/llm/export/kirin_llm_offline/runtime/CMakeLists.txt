cmake_minimum_required(VERSION 3.16)
project(kirin_full_npu_host LANGUAGES CXX)
get_filename_component(DEFAULT_MOBIINFER_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../.." ABSOLUTE)
set(MOBIINFER_ROOT "${DEFAULT_MOBIINFER_ROOT}" CACHE PATH "MobiInfer source directory")
set(MNN_HOST_LIBRARY "" CACHE FILEPATH "Existing host libMNN.so (tokenizer timing dependency)")
set(KIRIN_RUNTIME_PACKAGE "" CACHE PATH "Optional prepared full-NPU package for runtime failure tests")
if(NOT EXISTS "${MNN_HOST_LIBRARY}")
    message(FATAL_ERROR "Set MNN_HOST_LIBRARY to an existing host libMNN.so")
endif()
add_library(kirin_runtime STATIC FullNpuRuntime.cpp FullNpuCache.cpp
    ${MOBIINFER_ROOT}/transformers/llm/engine/src/tokenizer/tokenizer.cpp
    ${MOBIINFER_ROOT}/transformers/llm/engine/src/tokenizer/unicode.cpp
    ${MOBIINFER_ROOT}/transformers/llm/engine/src/tokenizer/unicode_data.cpp)
target_compile_features(kirin_runtime PUBLIC cxx_std_17)
target_compile_options(kirin_runtime PRIVATE -fvisibility=hidden -fno-rtti -fno-exceptions)
target_include_directories(kirin_runtime PUBLIC ${CMAKE_CURRENT_LIST_DIR}
    PRIVATE ${MOBIINFER_ROOT}/transformers/llm/engine/src ${MOBIINFER_ROOT}/3rd_party
            ${MOBIINFER_ROOT}/3rd_party/imageHelper ${MOBIINFER_ROOT}/include)
target_link_libraries(kirin_runtime PUBLIC ${MNN_HOST_LIBRARY})
add_executable(host_driver host_driver.cpp)
target_link_libraries(host_driver PRIVATE kirin_runtime)
add_executable(test_runtime test_runtime.cpp)
target_link_libraries(test_runtime PRIVATE kirin_runtime)
target_include_directories(test_runtime PRIVATE ${MOBIINFER_ROOT}/3rd_party)
add_executable(test_utf8 test_utf8.cpp)
target_compile_features(test_utf8 PRIVATE cxx_std_17)
target_compile_options(test_utf8 PRIVATE -UNDEBUG)
add_executable(test_cache test_cache.cpp FullNpuCache.cpp)
target_compile_features(test_cache PRIVATE cxx_std_17)
add_executable(test_bind test_bind.cpp FullNpuCache.cpp)
target_compile_features(test_bind PRIVATE cxx_std_17)
enable_testing()
add_test(NAME cache COMMAND test_cache)
add_test(NAME utf8_stream COMMAND test_utf8)
add_test(NAME port_binding COMMAND test_bind)
add_executable(test_transfer test_transfer.cpp FullNpuCache.cpp)
target_compile_features(test_transfer PRIVATE cxx_std_17)
target_compile_options(test_transfer PRIVATE -UNDEBUG -fno-rtti -fno-exceptions)
add_test(NAME incremental_kv_transfer COMMAND test_transfer)
add_executable(test_nnrt_loading test_nnrt/test_backend.cpp NnrtBackend.cpp FullNpuCache.cpp)
target_compile_features(test_nnrt_loading PRIVATE cxx_std_17)
target_compile_options(test_nnrt_loading PRIVATE -UNDEBUG -fno-rtti -fno-exceptions)
target_include_directories(test_nnrt_loading PRIVATE ${CMAKE_CURRENT_LIST_DIR}/test_nnrt ${CMAKE_CURRENT_LIST_DIR})
add_test(NAME nnrt_loading_call_substitutes COMMAND test_nnrt_loading)
target_link_libraries(test_nnrt_loading PRIVATE ${CMAKE_DL_LIBS})
set_target_properties(test_nnrt_loading PROPERTIES ENABLE_EXPORTS ON)
add_library(test_hiai_local SHARED test_nnrt/test_hiai_library.cpp)
set_target_properties(test_hiai_local PROPERTIES OUTPUT_NAME hiai_foundation
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/hiai_local")
add_library(test_hiai_missing SHARED test_nnrt/test_hiai_library.cpp)
target_compile_definitions(test_hiai_missing PRIVATE KIRIN_TEST_MISSING_INIT_WEIGHTS)
set_target_properties(test_hiai_missing PROPERTIES OUTPUT_NAME hiai_foundation
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/hiai_missing")
add_executable(test_nnrt_local test_nnrt/test_backend.cpp NnrtBackend.cpp FullNpuCache.cpp)
target_compile_features(test_nnrt_local PRIVATE cxx_std_17)
target_compile_options(test_nnrt_local PRIVATE -UNDEBUG -fno-rtti -fno-exceptions)
target_compile_definitions(test_nnrt_local PRIVATE KIRIN_TEST_LOCAL_HIAI)
target_include_directories(test_nnrt_local PRIVATE ${CMAKE_CURRENT_LIST_DIR}/test_nnrt ${CMAKE_CURRENT_LIST_DIR})
target_link_libraries(test_nnrt_local PRIVATE ${CMAKE_DL_LIBS})
set_target_properties(test_nnrt_local PROPERTIES ENABLE_EXPORTS ON)
add_dependencies(test_nnrt_local test_hiai_local test_hiai_missing)
add_test(NAME nnrt_local_library_symbol COMMAND ${CMAKE_COMMAND} -E env
    "LD_LIBRARY_PATH=$<TARGET_FILE_DIR:test_hiai_local>" $<TARGET_FILE:test_nnrt_local> local)
add_test(NAME nnrt_missing_library_symbol COMMAND ${CMAKE_COMMAND} -E env
    "LD_LIBRARY_PATH=$<TARGET_FILE_DIR:test_hiai_missing>" $<TARGET_FILE:test_nnrt_local> missing)
add_test(NAME nnrt_unavailable_library COMMAND ${CMAKE_COMMAND} -E env
    "LD_LIBRARY_PATH=${CMAKE_CURRENT_BINARY_DIR}/hiai_unavailable" $<TARGET_FILE:test_nnrt_local> unavailable)
if(EXISTS "${KIRIN_RUNTIME_PACKAGE}/config.json")
    add_test(NAME runtime_failures COMMAND test_runtime ${KIRIN_RUNTIME_PACKAGE}/config.json)
endif()
# Assertions are the test oracle even in a Release build.
target_compile_options(test_cache PRIVATE -UNDEBUG)
target_compile_options(test_bind PRIVATE -UNDEBUG)
target_compile_options(test_runtime PRIVATE -UNDEBUG)
