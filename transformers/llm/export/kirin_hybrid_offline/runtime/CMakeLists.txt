cmake_minimum_required(VERSION 3.16)
project(offline_npu2_host LANGUAGES CXX)
get_filename_component(REPO "${CMAKE_CURRENT_LIST_DIR}/../../../../.." ABSOLUTE)
set(MNN_HOST_LIBRARY "" CACHE FILEPATH "Host libMNN with CPU LLM/Omni enabled")
if(NOT EXISTS "${MNN_HOST_LIBRARY}")
    message(FATAL_ERROR "Set MNN_HOST_LIBRARY")
endif()
add_library(offline_npu2 STATIC OfflineNpu2.cpp Protocol.cpp Vision.cpp HostImage.cpp
    ${REPO}/transformers/llm/export/kirin_llm_offline/runtime/FullNpuCache.cpp)
target_compile_features(offline_npu2 PUBLIC cxx_std_17)
target_compile_options(offline_npu2 PRIVATE -fvisibility=hidden -fno-rtti)
target_include_directories(offline_npu2 PUBLIC ${CMAKE_CURRENT_LIST_DIR} ${REPO}/include
    ${REPO}/transformers/llm/engine/include
    ${REPO}/transformers/llm/export/kirin_llm_offline/runtime
    PRIVATE ${REPO}/transformers/llm/engine/src ${REPO}/3rd_party ${REPO}/3rd_party/imageHelper)
target_link_libraries(offline_npu2 PUBLIC ${MNN_HOST_LIBRARY})
get_filename_component(MNN_BUILD "${MNN_HOST_LIBRARY}" DIRECTORY)
if(EXISTS "${MNN_BUILD}/libllm.so")
    target_link_libraries(offline_npu2 PUBLIC ${MNN_BUILD}/libllm.so ${MNN_BUILD}/express/libMNN_Express.so)
endif()
add_executable(test_protocol test_protocol.cpp)
target_compile_options(test_protocol PRIVATE -UNDEBUG)
target_link_libraries(test_protocol PRIVATE offline_npu2)
add_executable(host_driver host_driver.cpp)
target_link_libraries(host_driver PRIVATE offline_npu2)
add_executable(mnn_host_driver mnn_host_driver.cpp)
target_include_directories(mnn_host_driver PRIVATE ${REPO}/3rd_party)
target_link_libraries(mnn_host_driver PRIVATE offline_npu2)
enable_testing()
add_test(NAME image_deepstack_mrope COMMAND test_protocol)
